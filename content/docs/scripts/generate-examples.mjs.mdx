import { writeFileSync } from 'node:fs';

const NOW = '2026-08-06T12:00:00.000Z';

const cell = (x, y, value, extra = {}) => ({
	x,
	y,
	type: 'TEXT',
	value: String(value),
	last_modified: NOW,
	...extra
});

const formula = (x, y, code, display, accessed) =>
	cell(x, y, display, {
		type: 'FORMULA',
		formula_code: code,
		evaluation_result: {
			success: true,
			output_value: String(display),
			cells_accessed: accessed,
			error_span: null
		}
	});

const fmt = (x, y, props) => ({ x, y, ...props });

// range helpers -> cells_accessed lists
const colRange = (x, y0, y1) => {
	const out = [];
	for (let y = y0; y <= y1; y++) out.push([x, y]);
	return out;
};

// ---------- date helpers (mirror the Rust core) ----------
const day = (iso) => Math.round(Date.parse(iso + 'T00:00:00Z') / 86400000);
const isoOf = (d) => new Date(d * 86400000).toISOString().slice(0, 10);
const weekdayIso = (d) => ((d + 3) % 7) + 1; // Mon=1..Sun=7
const networkdays = (a, b) => {
	let n = 0;
	for (let d = day(a); d <= day(b); d++) if (weekdayIso(d) < 6) n++;
	return n;
};
const isoWeek = (iso) => {
	const d = day(iso);
	const thu = d + 4 - weekdayIso(d);
	const y = new Date(thu * 86400000).getUTCFullYear();
	return Math.floor((thu - day(`${y}-01-01`)) / 7) + 1;
};
