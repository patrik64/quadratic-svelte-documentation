<script lang="ts">
	import { app } from '../state.svelte';
	import Modal from './Modal.svelte';

	interface CellTypeOption {
		name: string;
		mode?: 'FORMULA' | 'PYTHON' | 'JAVASCRIPT';
		color: string;
		description: string;
		disabled?: boolean;
		chip?: string;
	}

	const OPTIONS: CellTypeOption[] = [
		{
			name: 'Python',
			mode: 'PYTHON',
			color: 'var(--python)',
			description: 'Script with Pandas, NumPy, Micropip, and more.'
		},
		{
			name: 'Formula',
			mode: 'FORMULA',
			color: 'var(--formula)',
			description: 'Classic spreadsheet logic like SUM, AVERAGE, and more.'
		},
		{
			name: 'JavaScript',
			mode: 'JAVASCRIPT',
			color: 'var(--javascript)',
			description: 'Script with modern JavaScript; return a value to the sheet.'
		},
		{
			name: 'SQL Query',
			color: 'var(--faint-2)',
			description: 'Import your data with queries.',
			disabled: true,
			chip: 'Coming soon'
		}
	];

	let search = $state('');
	let selectedIndex = $state(0);
	let searchEl: HTMLInputElement | undefined = $state();

	const filtered = $derived(
		OPTIONS.filter((o) => o.name.toLowerCase().includes(search.toLowerCase()))
	);

	$effect(() => {
		searchEl?.focus();
	});
	$effect(() => {
		void search;
		selectedIndex = 0;
	});

	function close(): void {
		app.showCellTypeMenu = false;
	}

	function choose(option: CellTypeOption): void {
		if (option.disabled || !option.mode) return;
		app.editorMode = option.mode;
		app.editorCell = { ...app.cursorPosition };
		app.showCellTypeMenu = false;
		app.showCodeEditor = true;
	}

	function moveSelection(delta: number): void {
		if (filtered.length === 0) return;
		let i = selectedIndex;
		for (let step = 0; step < filtered.length; step++) {
			i = (i + delta + filtered.length) % filtered.length;
			if (!filtered[i].disabled) break;
		}
		selectedIndex = i;
	}

	function onKeyDown(e: KeyboardEvent): void {
		e.stopPropagation();
		if (e.key === 'Escape') close();
		else if (e.key === 'ArrowDown') {
			e.preventDefault();
			moveSelection(1);
		} else if (e.key === 'ArrowUp') {
			e.preventDefault();
			moveSelection(-1);
		} else if (e.key === 'Enter') {
			e.preventDefault();
			const option = filtered[selectedIndex];
			if (option) choose(option);
		}
	}
</script>

<Modal onclose={close}>
	<!-- svelte-ignore a11y_no_static_element_interactions -->
	<div onkeydown={onKeyDown}>
		<input
			bind:this={searchEl}
			bind:value={search}
			class="search"
			placeholder="Choose a cell type…"
			spellcheck="false"
		/>
		<div class="list">
			{#each filtered as option, i (option.name)}
				<button
					class="option"
					class:selected={i === selectedIndex}
					class:disabled={option.disabled}
					onclick={() => choose(option)}
					onmouseenter={() => {
						if (!option.disabled) selectedIndex = i;
					}}
				>
					<span class="icon" style:background={option.color}
						>{option.name.slice(0, 1)}</span
					>
					<span class="text">
						<span class="name">{option.name}</span>
						<span class="desc">{option.description}</span>
					</span>
					{#if option.chip}<span class="chip">{option.chip}</span>{/if}
				</button>
			{:else}
				<div class="empty">No matches</div>
			{/each}
		</div>
	</div>
</Modal>
