<script lang="ts">
	// Right-click grid menu, ported from the modern client's GridContextMenuCell.
	import { isCodeCellType } from '../core/types';
	import { KeyboardSymbols } from '../helpers/keyboardSymbols';
	import {
		app,
		copySelection,
		copySelectionAsPng,
		deleteColumns,
		deleteRows,
		downloadSelectionAsCsv,
		getSelectionRect,
		insertColumn,
		insertRow,
		pasteFromClipboard,
		rerunCurrentCell,
		sheetController
	} from '../state.svelte';

	const C = KeyboardSymbols.Cmd;
	const S = KeyboardSymbols.Shift;

	let { x, y, onclose }: { x: number; y: number; onclose: () => void } = $props();
	let menuEl: HTMLDivElement | undefined = $state();

	const sel = $derived.by(() => {
		void app.redraw;
		return getSelectionRect();
	});
	const cols = $derived(sel.x1 - sel.x0 + 1);
	const rows = $derived(sel.y1 - sel.y0 + 1);
	const isCodeCell = $derived.by(() => {
		void app.redraw;
		const cell = sheetController.sheet.getCell(app.cursorPosition.x, app.cursorPosition.y);
		return cell !== undefined && isCodeCellType(cell.type);
	});

	// keep the menu on-screen
	const left = $derived(Math.min(x, (typeof window !== 'undefined' ? window.innerWidth : 1200) - 250));
	const top = $derived(Math.min(y, (typeof window !== 'undefined' ? window.innerHeight : 800) - 420));

	function run(action: () => unknown): void {
		onclose();
		void action();
	}

	$effect(() => {
		// containment check rather than stopPropagation: Svelte delegates
		// mousedown, so stopping it inside the menu is unreliable
		const close = (e: MouseEvent) => {
			if (!menuEl?.contains(e.target as Node)) onclose();
		};
		const key = (e: KeyboardEvent) => {
			if (e.key === 'Escape') {
				e.stopPropagation();
				onclose();
			}
		};
		window.addEventListener('mousedown', close);
		window.addEventListener('keydown', key, true);
		return () => {
			window.removeEventListener('mousedown', close);
			window.removeEventListener('keydown', key, true);
		};
	});
</script>
