<script lang="ts">
	// Formula reference browser — the function list and docs come straight
	// from the Rust core's registry (formula_docs() wasm export).
	import { getFormulaDocs, type FormulaDoc } from '../formulas/runFormula';
	import { app } from '../state.svelte';
	import Modal from './Modal.svelte';

	let docs = $state<FormulaDoc[]>([]);
	let loading = $state(true);
	let search = $state('');
	let searchEl: HTMLInputElement | undefined = $state();

	$effect(() => {
		void getFormulaDocs()
			.then((d) => (docs = d))
			.finally(() => (loading = false));
	});
	$effect(() => {
		if (!loading) searchEl?.focus();
	});

	const filtered = $derived.by(() => {
		const q = search.trim().toLowerCase();
		if (!q) return docs;
		return docs.filter(
			(d) => d.name.toLowerCase().includes(q) || d.doc.toLowerCase().includes(q)
		);
	});

	const groups = $derived.by(() => {
		const map = new Map<string, FormulaDoc[]>();
		for (const doc of filtered) {
			const list = map.get(doc.category);
			if (list) list.push(doc);
			else map.set(doc.category, [doc]);
		}
		return [...map.entries()];
	});

	const canInsert = $derived(app.showCodeEditor && app.editorMode === 'FORMULA');

	function close(): void {
		app.showFormulaDocs = false;
	}

	function insert(doc: FormulaDoc): void {
		if (!canInsert) return;
		const takesArgs = !(doc.usages.length === 1 && doc.usages[0].endsWith('()'));
		app.pendingEditorInsert = doc.name + (takesArgs ? '(' : '()');
		close();
	}

	function onKeyDown(e: KeyboardEvent): void {
		e.stopPropagation();
		if (e.key === 'Escape') close();
	}
</script>

<Modal onclose={close} wide>
	<!-- svelte-ignore a11y_no_static_element_interactions -->
	<div class="docs" onkeydown={onKeyDown}>
		<div class="head">

				bind:this={searchEl}

					{#each functions as doc (doc.name)}
						<button class="fn" class:insertable={canInsert} onclick={() => insert(doc)}>

									{#each doc.examples as example (example)}<code>{example}</code>{/each}

						</button>
					{/each}
				{:else}
					<div class="empty">No functions match “{search}”</div>
				{/each}
			{/if}
		</div>
	</div>
</Modal>
