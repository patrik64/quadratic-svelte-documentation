const MAX_ARRAY_CELLS: i64 = 100_000;

type Row = SmallVec<[Value; 1]>;

/// Converts a value into rectangular rows (a scalar becomes 1×1).
fn to_rows(v: Spanned<Value>) -> FormulaResult<Vec<Row>> {
    let span = v.span;
    let rows = v.inner.into_array();
    let width = rows.first().map_or(0, |r| r.len());
    if rows.iter().any(|r| r.len() != width) {
        return Err(FormulaErrorMsg::NonRectangularArray.with_span(span));
    }
    Ok(rows)
}

/// Sort/equality classification: numbers < text < booleans, blanks last.
enum Key {
    Number(f64),
    Text(String),
    Bool(bool),
    Blank,
}

fn classify(v: &Value) -> Key {
    match v {
        Value::Bool(b) => Key::Bool(*b),
        Value::Number(n) => Key::Number(*n),
        Value::String(s) if s.trim().is_empty() => Key::Blank,
        other => match other.to_number() {
            Ok(n) => Key::Number(n),
            Err(_) => Key::Text(other.to_string().to_lowercase()),
        },
    }
}

fn compare_values(a: &Value, b: &Value) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    let rank = |k: &Key| match k {
        Key::Number(_) => 0,
        Key::Text(_) => 1,
        Key::Bool(_) => 2,
        Key::Blank => 3, // blanks sort last in both directions (handled by caller)
    };
    let (ka, kb) = (classify(a), classify(b));
    match rank(&ka).cmp(&rank(&kb)) {
        Ordering::Equal => match (ka, kb) {
            (Key::Number(x), Key::Number(y)) => x.partial_cmp(&y).unwrap_or(Ordering::Equal),
            (Key::Text(x), Key::Text(y)) => x.cmp(&y),
            (Key::Bool(x), Key::Bool(y)) => x.cmp(&y),
            _ => Ordering::Equal,
        },
        other => other,
    }
}

fn values_equivalent(a: &Value, b: &Value) -> bool {
    match (classify(a), classify(b)) {
        (Key::Number(x), Key::Number(y)) => x == y,
        (Key::Text(x), Key::Text(y)) => x == y,
        (Key::Bool(x), Key::Bool(y)) => x == y,
        (Key::Blank, Key::Blank) => true,
        _ => false,
    }
}

/// Truthiness for FILTER's include mask: booleans, and numbers where 0 is
/// false (matching how comparison results and cell data arrive here).
fn truthy(v: &Spanned<Value>) -> FormulaResult<bool> {
    if let Ok(b) = v.inner.to_bool() {
        return Ok(b);
    }
    match &v.inner {
        Value::String(s) if s.trim().is_empty() => Ok(false),
        other => match other.to_number() {
            Ok(n) => Ok(n != 0.0),
            Err(_) => Err(FormulaErrorMsg::Expected {
                expected: "boolean or number in the include mask".into(),
                got: Some(other.type_name().into()),
            }
            .with_span(v.span)),
        },
    }
}

#[cfg(test)]
mod tests {
    use crate::formulas::tests::*;

    fn eval(s: &str) -> String {
        eval_to_string(&mut NoGrid, s)
    }

    #[test]
    fn test_transpose() {
        assert_eq!("{1, 3; 2, 4}", eval("TRANSPOSE({1, 2; 3, 4})"));
        assert_eq!("{1; 2; 3}", eval("TRANSPOSE({1, 2, 3})"));
    }

    #[test]
    fn test_sequence() {
        assert_eq!("{1; 2; 3}", eval("SEQUENCE(3)"));
        assert_eq!("{0, 10, 20; 30, 40, 50}", eval("SEQUENCE(2, 3, 0, 10)"));
    }

    #[test]
    fn test_unique() {
        assert_eq!("{1; 2; 3}", eval("UNIQUE({1; 2; 1; 3; 2})"));
        // case-insensitive text, numeric-string awareness
        assert_eq!("{a; b}", eval("UNIQUE({'a'; 'A'; 'b'})"));
        assert_eq!("{1}", eval("UNIQUE({1; '1'})"));
    }

    #[test]
    fn test_sort() {
        assert_eq!("{1; 2; 3}", eval("SORT({3; 1; 2})"));
        assert_eq!("{3; 2; 1}", eval("SORT({3; 1; 2}, 1, -1)"));
        // sort rows by second column
        assert_eq!("{b, 1; a, 2}", eval("SORT({'a', 2; 'b', 1}, 2)"));
        // numbers before text, blanks last even descending
        assert_eq!("{5; x; }", eval("SORT({'x'; ''; 5})"));
        assert_eq!("{x; 5; }", eval("SORT({'x'; ''; 5}, 1, -1)"));
    }

    #[test]
    fn test_filter() {
        // the old parser's array literals only take atoms, so masks are
        // numeric here; real ranges deliver booleans, which truthy() accepts
        assert_eq!("{1; 3}", eval("FILTER({1; 2; 3}, {1; 0; 1})"));
        assert_eq!("{2, 20}", eval("FILTER({1, 10; 2, 20}, {0; 1})"));
        assert_eq!("none", eval("FILTER({1; 2}, {0; 0}, 'none')"));
    }

    #[test]
    fn test_sumproduct() {
        assert_eq!("32", eval("SUMPRODUCT({1; 2; 3}, {4; 5; 6})"));
        assert_eq!("6", eval("SUMPRODUCT({1, 2, 3})"));
    }
}
