/// Helper for LEFT/RIGHT: `(string)` or `(string, count)`, count defaults to 1.
fn string_and_optional_count(
    args: Spanned<Vec<Spanned<Value>>>,
) -> FormulaResult<(String, usize)> {
    let span = args.span;
    let mut a = args.inner.into_iter();
    let (Some(string), count) = (a.next(), a.next()) else {
        return Err(FormulaErrorMsg::BadArgumentCount.with_span(span));
    };
    if a.next().is_some() {
        return Err(FormulaErrorMsg::BadArgumentCount.with_span(span));
    }
    let count = match count {
        Some(c) => to_count(c.to_number()?),
        None => 1,
    };
    Ok((string.to_string(), count))
}
