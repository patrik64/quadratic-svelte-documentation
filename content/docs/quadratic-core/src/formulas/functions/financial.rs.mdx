// ---------- shared helpers (ported verbatim from the modern core) ----------

fn calculate_pmt(rate: f64, nper: f64, pv: f64, fv: f64, payment_type: f64) -> f64 {
    if rate == 0.0 {
        -(pv + fv) / nper
    } else {
        let pvif = (1.0 + rate).powf(nper);
        let pmt = rate * (pv * pvif + fv) / (pvif - 1.0);
        -pmt / (1.0 + rate * payment_type)
    }
}

fn calculate_fv(rate: f64, nper: f64, pmt: f64, pv: f64, payment_type: f64) -> f64 {
    if rate == 0.0 {
        -(pv + pmt * nper)
    } else {
        let pvif = (1.0 + rate).powf(nper);
        let fvifa = (pvif - 1.0) / rate;
        -pv * pvif - pmt * (1.0 + rate * payment_type) * fvifa
    }
}

fn calculate_pv(rate: f64, nper: f64, pmt: f64, fv: f64, payment_type: f64) -> f64 {
    if rate == 0.0 {
        -(fv + pmt * nper)
    } else {
        let pvif = (1.0 + rate).powf(nper);
        let fvifa = (pvif - 1.0) / rate;
        (-fv - pmt * (1.0 + rate * payment_type) * fvifa) / pvif
    }
}

fn normalize_payment_type(payment_type: Option<f64>) -> f64 {
    if payment_type.unwrap_or(0.0) != 0.0 { 1.0 } else { 0.0 }
}

/// Bisection fallback for the rate solvers: scans (-0.9999, 10) for a sign
/// change of `f`, then halves the bracket. Returns None when no root is
/// bracketed.
fn bisect(f: impl Fn(f64) -> f64) -> Option<f64> {
    const STEPS: usize = 400;
    let x = |i: usize| -0.9999 + (10.0 + 0.9999) * i as f64 / STEPS as f64;
    let mut lo = x(0);
    let mut flo = f(lo);
    let mut bracket = None;
    for i in 1..=STEPS {
        let hi = x(i);
        let fhi = f(hi);
        if flo.is_finite() && fhi.is_finite() && flo * fhi <= 0.0 {
            bracket = Some((lo, hi));
            break;
        }
        lo = hi;
        flo = fhi;
    }
    let (mut lo, mut hi) = bracket?;
    let mut flo = f(lo);
    for _ in 0..200 {
        let mid = (lo + hi) / 2.0;
        let fm = f(mid);
        if !fm.is_finite() {
            return None;
        }
        if flo * fm <= 0.0 {
            hi = mid;
        } else {
            lo = mid;
            flo = fm;
        }
    }
    Some((lo + hi) / 2.0)
}

/// Amortization walk shared by IPMT/PPMT/CUMIPMT/CUMPRINC: returns the
/// balance entering `per` given the standard payment.
fn balance_before(rate: f64, pv: f64, pmt: f64, payment_type: f64, per: i64) -> f64 {
    let mut balance = pv;
    for _ in 1..per {
        if payment_type == 1.0 {
            balance += pmt;
        }
        let interest = balance * rate;
        balance = balance + interest + pmt;
        if payment_type == 1.0 {
            balance -= pmt;
        }
    }
    balance
}

            usages: &["values", "values, guess"],
            examples: &["IRR({-10000, 3000, 4200, 6800})"],
            doc: "Internal rate of return of periodic cash flows, solved \
                  iteratively.",
            eval: util::pure_fn(|args| {
                let a = Args::new(args, 1, 2)?;
                let cash_flows = a.numbers_of(0);
                let guess = a.opt_num(1)?.unwrap_or(0.1);
                if cash_flows.len() < 2
                    || !cash_flows.iter().any(|&v| v > 0.0)
                    || !cash_flows.iter().any(|&v| v < 0.0)
                {
                    return Err(a.invalid());
                }

                let npv_at = |rate: f64| {
                    cash_flows
                        .iter()
                        .enumerate()
                        .map(|(i, &cf)| cf / (1.0 + rate).powi(i as i32))
                        .sum::<f64>()
                };

                let mut rate = guess;
                for _ in 0..100 {
                    let mut npv = 0.0;
                    let mut dnpv = 0.0;
                    for (i, &cf) in cash_flows.iter().enumerate() {
                        let factor = (1.0 + rate).powi(i as i32);
                        npv += cf / factor;
                        dnpv -= (i as f64) * cf / factor / (1.0 + rate);
                    }
                    if dnpv.abs() < 1e-15 {
                        break;
                    }
                    let new_rate = rate - npv / dnpv;
                    if (new_rate - rate).abs() < 1e-10 {
                        rate = new_rate;
                        break;
                    }
                    rate = new_rate;
                }

                if !rate.is_finite() || rate <= -1.0 {
                    rate = bisect(npv_at).ok_or_else(|| a.invalid())?;
                }
                Ok(Value::Number(rate))
            }),
        },
        FormulaFunction {
            name: "MIRR",
            arg_completion: "${1:values}, ${2:finance_rate}, ${3:reinvest_rate}",
            usages: &["values, finance_rate, reinvest_rate"],
            examples: &["MIRR({-10000, 3000, 4200, 6800}, 0.1, 0.12)"],
            doc: "Modified internal rate of return, with separate rates for \
                  financing and reinvestment.",
            eval: util::pure_fn(|args| {
                let a = Args::new(args, 3, 3)?;
                let cash_flows = a.numbers_of(0);
                let (finance_rate, reinvest_rate) = (a.num(1)?, a.num(2)?);
                let n = cash_flows.len();
                if n < 2 {
                    return Err(a.invalid());
                }
                let mut npv_neg = 0.0;
                let mut fv_pos = 0.0;
                for (i, &cf) in cash_flows.iter().enumerate() {
                    if cf < 0.0 {
                        npv_neg += cf / (1.0 + finance_rate).powi(i as i32);
                    } else {
                        fv_pos += cf * (1.0 + reinvest_rate).powi((n - 1 - i) as i32);
                    }
                }
                if npv_neg == 0.0 || fv_pos == 0.0 {
                    return Err(a.invalid());
                }
                Ok(Value::Number((-fv_pos / npv_neg).powf(1.0 / (n - 1) as f64) - 1.0))
            }),
        },
        FormulaFunction {
            name: "XNPV",
            arg_completion: "${1:rate}, ${2:values}, ${3:dates}",
            usages: &["rate, values, dates"],
            examples: &["XNPV(0.1, {-10000, 3000}, {\"2026-01-01\", \"2026-06-01\"})"],
            doc: "Net present value of cash flows on arbitrary dates.",
            eval: util::pure_fn(|args| {
                let a = Args::new(args, 3, 3)?;
                let rate = a.num(0)?;
                let values = a.numbers_of(1);
                let days = a.days_of(2)?;
                if values.len() != days.len() || values.is_empty() {
                    return Err(a.invalid());
                }
                let d0 = days[0];
                let xnpv = values
                    .iter()
                    .zip(&days)
                    .map(|(&v, &d)| v / (1.0 + rate).powf((d - d0) as f64 / 365.0))
                    .sum::<f64>();
                Ok(Value::Number(xnpv))
            }),
        },
        FormulaFunction {
            name: "XIRR",
            arg_completion: "${1:values}, ${2:dates}",
            usages: &["values, dates", "values, dates, guess"],
            examples: &["XIRR({-10000, 12000}, {\"2026-01-01\", \"2026-12-31\"})"],
            doc: "Internal rate of return of cash flows on arbitrary dates, \
                  solved iteratively.",
            eval: util::pure_fn(|args| {
                let a = Args::new(args, 2, 3)?;
                let values = a.numbers_of(0);
                let days = a.days_of(1)?;
                let guess = a.opt_num(2)?.unwrap_or(0.1);
                if values.len() != days.len() || values.is_empty() {
                    return Err(a.invalid());
                }
                let d0 = days[0];
                let flows: Vec<(f64, f64)> = values
                    .iter()
                    .zip(&days)
                    .map(|(&v, &d)| (v, (d - d0) as f64 / 365.0))
                    .collect();

                let xnpv_at = |rate: f64| {
                    flows.iter().map(|&(cf, years)| cf / (1.0 + rate).powf(years)).sum::<f64>()
                };

                let mut rate = guess;
                for _ in 0..100 {
                    let mut xnpv = 0.0;
                    let mut dxnpv = 0.0;
                    for &(cf, years) in &flows {
                        let factor = (1.0 + rate).powf(years);
                        xnpv += cf / factor;
                        if years != 0.0 {
                            dxnpv -= years * cf / factor / (1.0 + rate);
                        }
                    }
                    if dxnpv.abs() < 1e-15 {
                        break;
                    }
                    let new_rate = rate - xnpv / dxnpv;
                    if (new_rate - rate).abs() < 1e-10 {
                        rate = new_rate;
                        break;
                    }
                    rate = new_rate;
                }

                if !rate.is_finite() || rate <= -1.0 {
                    rate = bisect(xnpv_at).ok_or_else(|| a.invalid())?;
                }
                Ok(Value::Number(rate))
            }),
        },
        FormulaFunction {
            name: "FVSCHEDULE",
            arg_completion: "${1:principal}, ${2:schedule}",
            usages: &["principal, schedule"],
            examples: &["FVSCHEDULE(1000, {0.09, 0.11, 0.1})"],
            doc: "Future value of a principal after a schedule of compound \
                  interest rates.",
            eval: util::pure_fn(|args| {
                let a = Args::new(args, 2, 2)?;
                let mut fv = a.num(0)?;
                for rate in a.numbers_of(1) {
                    fv *= 1.0 + rate;
                }
                Ok(Value::Number(fv))
            }),
        },
    ]
}

#[cfg(test)]
mod tests {
    use crate::formulas::tests::*;

    fn eval_num(s: &str) -> f64 {
        eval_to_string(&mut NoGrid, s).parse().unwrap()
    }

    #[test]
    fn test_pmt_fv_pv() {
        // canonical Excel values
        let pmt = eval_num("PMT(0.08/12, 60, 10000)");
        assert!((pmt - -202.7639429).abs() < 1e-4, "PMT was {pmt}");
        let fv = eval_num("FV(0.08/12, 60, -200)");
        assert!((fv - 14695.3712).abs() < 1e-3, "FV was {fv}");
        let pv = eval_num("PV(0.08/12, 60, -200)");
        assert!((pv - 9863.6867).abs() < 1e-3, "PV was {pv}");
        assert_eq!(-200.0, eval_num("PMT(0, 50, 10000)"));
    }

    #[test]
    fn test_nper_rate() {
        assert!((eval_num("NPER(0.08/12, -200, 10000)") - 61.02).abs() < 0.1);
        // RATE(60, -202.7639446, 10000) should recover 0.08/12
        assert!((eval_num("RATE(60, -202.7639446, 10000)") - 0.0066666667).abs() < 1e-8);
    }

    #[test]
    fn test_ipmt_ppmt() {
        let ipmt = eval_num("IPMT(0.08/12, 1, 60, 10000)");
        let ppmt = eval_num("PPMT(0.08/12, 1, 60, 10000)");
        assert!((ipmt - -66.6666667).abs() < 1e-6);
        // interest + principal = the payment
        assert!((ipmt + ppmt - eval_num("PMT(0.08/12, 60, 10000)")).abs() < 1e-9);
        assert!((eval_num("ISPMT(0.08/12, 1, 12, 10000)") - -61.1111111).abs() < 1e-6);
    }

    #[test]
    fn test_cumulative() {
        let cumipmt = eval_num("CUMIPMT(0.08/12, 60, 10000, 1, 12, 0)");
        assert!(cumipmt < 0.0, "CUMIPMT returns a negative cash outflow");
        assert!((cumipmt - -738.766).abs() < 0.01, "CUMIPMT was {cumipmt}");
        let cumprinc = eval_num("CUMPRINC(0.08/12, 60, 10000, 1, 60, 0)");
        // over the full life, all principal is repaid
        assert!((cumprinc - -10000.0).abs() < 1e-6);
    }

    #[test]
    fn test_depreciation() {
        assert_eq!(90.0, eval_num("SLN(1000, 100, 10)"));
        assert!((eval_num("SYD(1000, 100, 10, 1)") - 163.6363636).abs() < 1e-6);
        assert_eq!(200.0, eval_num("DDB(1000, 100, 10, 1)"));
        assert!((eval_num("DB(1000, 100, 10, 1)") - 206.0).abs() < 0.01);
        assert!((eval_num("VDB(1000, 100, 10, 0, 1)") - 200.0).abs() < 1e-9);
    }

    #[test]
    fn test_npv_irr() {
        let npv = eval_num("NPV(0.1, {-10000, 3000, 4200, 6800})");
        assert!((npv - 1188.44).abs() < 0.5, "NPV was {npv}");
        let irr = eval_num("IRR({-10000, 3000, 4200, 6800})");
        // NPV at IRR must be ~0
        let check = -10000.0 + 3000.0 / (1.0 + irr) + 4200.0 / (1.0 + irr).powi(2)
            + 6800.0 / (1.0 + irr).powi(3);
        assert!(check.abs() < 1e-6);
        let mirr = eval_num("MIRR({-10000, 3000, 4200, 6800}, 0.1, 0.12)");
        // hand-check: FV+ at 12% = 15267.2 → (1.52672)^(1/3) − 1
        assert!((mirr - 0.151471).abs() < 1e-5, "MIRR was {mirr}");
    }

    #[test]
    fn test_xnpv_xirr() {
        let xirr = eval_num(
            "XIRR({-10000; 12000}, {\"2026-01-01\"; \"2026-12-31\"})"
        );
        // ~20% over almost exactly one year
        assert!((xirr - 0.2).abs() < 0.01, "XIRR was {xirr}");
        let xnpv = eval_num(
            "XNPV(0.1, {-10000; 11000}, {\"2026-01-01\"; \"2027-01-01\"})"
        );
        // 11000 discounted one year at 10% ≈ 10000 → XNPV ≈ -3
        assert!(xnpv.abs() < 10.0, "XNPV was {xnpv}");
        let fvs = eval_num("FVSCHEDULE(1000, {0.09, 0.11, 0.1})");
        assert!((fvs - 1331.0).abs() < 1.0, "FVSCHEDULE was {fvs}");
    }
}
